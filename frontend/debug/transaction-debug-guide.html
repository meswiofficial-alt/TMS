🖥️ Category 3: Frontend Validation Logic

// Located in app.js (line 108-129)
function collectFormData() {
    const inputs = document.querySelectorAll('#modalContainer input, #modalContainer select, #modalContainer textarea');
    const data = {};
    let valid = true;
    
    inputs.forEach(input => {
        const name = input.getAttribute('data-field');
        if (name) {
            // Check required validation
            if (input.hasAttribute('required') && !input.value.trim()) {
                valid = false;
                input.style.borderColor = '#ef4444';
            }
            let value = input.value.trim();
            if (input.type === 'number') value = parseFloat(value) || 0;
            data[name] = value;
        }
    });
    
    return valid ? data : null;  // Returns null if validation fails
}

// Run in browser console to test all buttons:
function testAllButtons() {
    const buttons = document.querySelectorAll('button, .btn, .action-btn, [onclick]');
    const results = [];
    
    buttons.forEach((btn, i) => {
        const test = {
            index: i,
            text: btn.textContent.trim().substring(0, 30),
            hasClick: btn.onclick !== null || btn.getAttribute('onclick') !== null,
            isDisabled: btn.disabled,
            visible: btn.offsetParent !== null
        };
        results.push(test);
        
        // Visual feedback
        if (test.hasClick && !test.isDisabled && test.visible) {
            btn.style.outline = '2px solid #22c55e';
        } else {
            btn.style.outline = '2px solid #ef4444';
        }
    });
    
    console.table(results);
    console.log(`Total buttons: ${buttons.length}`);
    console.log(`Clickable: ${results.filter(r => r.hasClick && !r.isDisabled).length}`);
    console.log(`Disabled: ${results.filter(r => r.isDisabled).length}`);
    console.log(`Hidden: ${results.filter(r => !r.visible).length}`);
    
    // Remove outlines after 3 seconds
    setTimeout(() => buttons.forEach(b => b.style.outline = ''), 3000);
    
    return results;
}

// Run the test
testAllButtons();
